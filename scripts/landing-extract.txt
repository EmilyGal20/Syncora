    <script>
      (() => {
        const reduced = window.matchMedia("(prefers-reduced-motion: reduce)").matches;
        const html = document.documentElement;
        const hero = document.querySelector(".hero");
        const heroHit = document.querySelector(".hero-mark-hit");
        const chevron = document.querySelector(".hero-scroll-hint");
        const canvas = document.querySelector(".hero-plus-grid");
        html.classList.add("motion-ready");

        const mark = (el) => el && el.classList.add("is-in");
        const steps = ["logo", "pills", "title", "lead", "chevron"];
        if (reduced) {
          hero.classList.add("is-entered", ...steps.map((s) => "step-" + s));
          document.querySelectorAll(".reveal").forEach(mark);
        } else {
          steps.forEach((step, index) => {
            window.setTimeout(() => {
              hero.classList.add("step-" + step);
              mark(hero.querySelector('[data-hero-step="' + step + '"]'));
              if (index === steps.length - 1) hero.classList.add("is-entered");
            }, 160 + index * 180);
          });
          const targets = Array.from(document.querySelectorAll(".reveal")).filter((el) => !el.closest(".hero"));
          const io = new IntersectionObserver((entries) => {
            entries.forEach((entry) => {
              if (!entry.isIntersecting) return;
              mark(entry.target);
              io.unobserve(entry.target);
            });
          }, { threshold: 0.12 });
          targets.forEach((el) => io.observe(el));
        }

        if (heroHit && !reduced) {
          const logo = heroHit.querySelector(".logo-bars");
          heroHit.addEventListener("click", () => {
            if (!logo) return;
            logo.classList.remove("is-waving");
            void logo.getBoundingClientRect();
            logo.classList.add("is-waving");
          });
        }

        document.querySelectorAll(".feature-card").forEach((card) => {
          const setPressed = (on) => card.classList.toggle("is-pressed", on);
          card.addEventListener("pointerdown", () => setPressed(true));
          card.addEventListener("pointerup", () => setPressed(false));
          card.addEventListener("pointerleave", () => setPressed(false));
          card.addEventListener("pointercancel", () => setPressed(false));
        });

        const onScroll = () => {
          if (!chevron || reduced) return;
          const bottom = hero.getBoundingClientRect().bottom;
          chevron.classList.toggle("is-gone", bottom < 48);
          chevron.classList.toggle("is-faded", bottom >= 48 && window.scrollY > 28);
        };
        window.addEventListener("scroll", onScroll, { passive: true });

        if (canvas && hero) {
          const ctx = canvas.getContext("2d");
          let cols = 0, rows = 0, gapX = 0, gapY = 0, width = 0, height = 0;
          let energy = null;
          let raf = 0;
          const layout = () => {
            const rect = hero.getBoundingClientRect();
            const dpr = Math.min(window.devicePixelRatio || 1, 2);
            width = Math.max(1, rect.width);
            height = Math.max(1, rect.height);
            canvas.width = Math.floor(width * dpr);
            canvas.height = Math.floor(height * dpr);
            ctx.setTransform(dpr, 0, 0, dpr, 0, 0);
            const target = width < 400 ? 34 : 40;
            cols = Math.max(6, Math.floor(width / target));
            rows = Math.max(8, Math.floor(height / target));
            gapX = width / (cols + 1);
            gapY = height / (rows + 1);
            energy = new Float32Array(cols * rows);
          };
          const draw = () => {
            ctx.clearRect(0, 0, width, height);
            ctx.font = "700 11px Segoe UI, Arial, sans-serif";
            ctx.textAlign = "center";
            ctx.textBaseline = "middle";
            let active = false;
            for (let r = 0; r < rows; r += 1) {
              for (let c = 0; c < cols; c += 1) {
                const i = r * cols + c;
                let e = energy[i];
                if (e > 0.001) {
                  e = Math.max(0, e - 0.02);
                  energy[i] = e;
                  active = active || e > 0.001;
                }
                const a = Math.min(0.85, 0.16 + e * 0.7);
                ctx.fillStyle = "rgba(46, 125, 160, " + a + ")";
                ctx.fillText("+", gapX + c * gapX, gapY + r * gapY);
              }
            }
            raf = active ? requestAnimationFrame(draw) : 0;
          };
          const ripple = (clientX, clientY) => {
            if (reduced || !energy) return;
            const rect = canvas.getBoundingClientRect();
            const x = clientX - rect.left;
            const y = clientY - rect.top;
            const radius = Math.max(gapX, gapY) * 2.4;
            for (let r = 0; r < rows; r += 1) {
              for (let c = 0; c < cols; c += 1) {
                const cx = gapX + c * gapX;
                const cy = gapY + r * gapY;
                const dist = Math.hypot(cx - x, cy - y);
                if (dist > radius) continue;
                const t = 1 - dist / radius;
                const i = r * cols + c;
                energy[i] = Math.min(1, energy[i] + t * t * 0.95);
              }
            }
            if (!raf) raf = requestAnimationFrame(draw);
          };
          layout();
          draw();
          canvas.addEventListener("pointerdown", (event) => {
            if (event.target.closest(".hero-mark-hit")) return;
            ripple(event.clientX, event.clientY);
          });
          window.addEventListener("resize", () => { layout(); if (!raf) raf = requestAnimationFrame(draw); });
        }
      })();
    </script>
